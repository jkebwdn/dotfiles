cmake_minimum_required(VERSION 3.25)
project(magi_lock_protocol_proof VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAGI_LOCK_DEVELOPMENT_UNLOCK "Compile UNSAFE timed development unlock" OFF)
option(MAGI_LOCK_FAILURE_INJECTION "Compile synthetic crash/hang selection; NEVER deploy" OFF)
if(MAGI_LOCK_FAILURE_INJECTION AND NOT MAGI_LOCK_DEVELOPMENT_UNLOCK)
    message(FATAL_ERROR "Failure injection requires development emergency recovery")
endif()
set(MAGI_LOCK_AUTH_WORKER "" CACHE FILEPATH "Explicit absolute prebuilt Rust worker; empty uses packaged sibling")
if(MAGI_LOCK_AUTH_WORKER AND NOT IS_ABSOLUTE "${MAGI_LOCK_AUTH_WORKER}")
    message(FATAL_ERROR "Auth worker path must be absolute")
endif()
set(MAGI_LOCK_AUTH_FIXTURE "" CACHE FILEPATH "Prebuilt test-only Rust protocol_fixture example")
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick Qml)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd>=252)
pkg_check_modules(XKB REQUIRED IMPORTED_TARGET xkbcommon>=1.0)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client>=1.20)
pkg_check_modules(PROTOCOLS REQUIRED wayland-protocols>=1.25)
pkg_get_variable(PROTOCOL_DIR wayland-protocols pkgdatadir)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(LOCK_XML "${PROTOCOL_DIR}/staging/ext-session-lock/ext-session-lock-v1.xml")
if(NOT EXISTS "${LOCK_XML}")
    message(FATAL_ERROR "Installed ext-session-lock-v1.xml is required")
endif()
add_custom_command(OUTPUT session-lock.h session-lock.c
    COMMAND "${WAYLAND_SCANNER}" client-header "${LOCK_XML}" session-lock.h
    COMMAND "${WAYLAND_SCANNER}" private-code "${LOCK_XML}" session-lock.c
    DEPENDS "${LOCK_XML}" VERBATIM)
add_library(lock_protocol STATIC "${CMAKE_CURRENT_BINARY_DIR}/session-lock.c")
target_include_directories(lock_protocol PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(lock_protocol PUBLIC PkgConfig::WAYLAND)
add_library(lock_auth STATIC src/credential.cpp src/auth_controller.cpp src/native_input.cpp)
target_include_directories(lock_auth PUBLIC src)
target_link_libraries(lock_auth PUBLIC Qt6::Core PkgConfig::XKB)
target_compile_options(lock_auth PRIVATE -Wall -Wextra -Wpedantic -Werror)
function(lock_core target development)
    add_library(${target} STATIC src/controller.cpp src/options.cpp src/sleep_policy.cpp src/instance_guard.cpp)
    target_include_directories(${target} PUBLIC src)
    target_link_libraries(${target} PUBLIC lock_auth)
    target_compile_definitions(${target} PUBLIC MAGI_LOCK_DEVELOPMENT_UNLOCK=${development} MAGI_LOCK_FAILURE_INJECTION=$<AND:$<BOOL:${MAGI_LOCK_FAILURE_INJECTION}>,$<BOOL:${development}>>)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endfunction()
if(MAGI_LOCK_DEVELOPMENT_UNLOCK)
    message(WARNING "UNSAFE DEVELOPMENT BUILD: explicitly enabled timed unlock; never deploy")
    lock_core(lock_core 1)
else()
    lock_core(lock_core 0)
endif()
add_library(lock_presentation STATIC src/presentation.cpp)
target_include_directories(lock_presentation PUBLIC src)
target_link_libraries(lock_presentation PUBLIC Qt6::Quick Qt6::Qml lock_auth)
qt_add_resources(lock_presentation proof_qml PREFIX "/magi-lock" FILES qml/Proof.qml)
add_executable(magi-lock-proof src/main.cpp src/wayland_owner.cpp src/logind_sleep.cpp)
target_link_libraries(magi-lock-proof PRIVATE lock_core lock_presentation lock_protocol Qt6::Gui PkgConfig::SYSTEMD)
target_compile_definitions(magi-lock-proof PRIVATE MAGI_LOCK_AUTH_WORKER="${MAGI_LOCK_AUTH_WORKER}")
target_compile_options(magi-lock-proof PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MAGI_LOCK_FAILURE_INJECTION)
    if(NOT IS_ABSOLUTE "${MAGI_LOCK_AUTH_FIXTURE}" OR NOT EXISTS "${MAGI_LOCK_AUTH_FIXTURE}")
        message(FATAL_ERROR "Explicit prebuilt failure fixture required")
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/failure-fixtures")
    foreach(mode crash hang)
        file(CREATE_LINK "${MAGI_LOCK_AUTH_FIXTURE}" "${CMAKE_CURRENT_BINARY_DIR}/failure-fixtures/fixture-${mode}" SYMBOLIC)
    endforeach()
    target_compile_definitions(magi-lock-proof PRIVATE MAGI_LOCK_FAILURE_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}/failure-fixtures")
endif()
# No install rule: this is not a production locker.
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME handoff_synthetic COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/handoff_test.py")
    add_test(NAME request_synthetic COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/request_test.py")
    set_tests_properties(handoff_synthetic request_synthetic PROPERTIES
        ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1" TIMEOUT 15)
    if(NOT IS_ABSOLUTE "${MAGI_LOCK_AUTH_FIXTURE}" OR NOT EXISTS "${MAGI_LOCK_AUTH_FIXTURE}")
        message(FATAL_ERROR "Build the Rust protocol_fixture example offline and set MAGI_LOCK_AUTH_FIXTURE for tests")
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    foreach(mode success failed pamerror crash exitonly hang version truncated oversize unknown wrongid zero-success echoon secondprompt badexit extraframe duplicate success-crash delayed-success eof-handshake eof-prompt eof-response result-hang protocolerror workererror cancelled)
        file(CREATE_LINK "${MAGI_LOCK_AUTH_FIXTURE}" "${CMAKE_CURRENT_BINARY_DIR}/fixtures/fixture-${mode}" SYMBOLIC)
    endforeach()
    lock_core(lock_core_production_test 0)
    lock_core(lock_core_development_test 1)
    foreach(mode production development)
        add_executable(controller_${mode}_test tests/controller_test.cpp)
        target_link_libraries(controller_${mode}_test PRIVATE lock_core_${mode}_test)
        add_test(NAME controller_${mode} COMMAND controller_${mode}_test)
        add_executable(auth_${mode}_test tests/auth_test.cpp)
        target_link_libraries(auth_${mode}_test PRIVATE lock_core_${mode}_test)
        target_compile_options(auth_${mode}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
        add_test(NAME auth_${mode} COMMAND auth_${mode}_test "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
        set_tests_properties(auth_${mode} PROPERTIES TIMEOUT 60)
    endforeach()
    add_executable(sleep_test tests/sleep_test.cpp)
    target_link_libraries(sleep_test PRIVATE lock_core_production_test)
    target_compile_options(sleep_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME sleep_synthetic COMMAND sleep_test "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    set_tests_properties(sleep_synthetic PROPERTIES TIMEOUT 45)
    add_executable(output_test tests/output_test.cpp)
    target_link_libraries(output_test PRIVATE lock_presentation lock_core_production_test)
    target_compile_options(output_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME output_synthetic COMMAND output_test "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    set_tests_properties(output_synthetic PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 60)
    add_executable(presentation_test tests/presentation_test.cpp)
    target_link_libraries(presentation_test PRIVATE lock_presentation lock_core_production_test)
    add_test(NAME presentation_offscreen COMMAND presentation_test "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    set_tests_properties(presentation_offscreen PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 30)
    add_test(NAME presentation_boundary COMMAND "${CMAKE_COMMAND}"
        -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/presentation_boundary.cmake")
    add_test(NAME binary_gate COMMAND "${CMAKE_COMMAND}"
        -DBINARY=$<TARGET_FILE:magi-lock-proof>
        -DDEVELOPMENT=${MAGI_LOCK_DEVELOPMENT_UNLOCK}
        -DINJECTION=${MAGI_LOCK_FAILURE_INJECTION}
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/binary_gate.cmake")
endif()
