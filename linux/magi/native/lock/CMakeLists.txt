function(lock_core target development)
    add_library(${target} STATIC src/controller.cpp src/options.cpp src/sleep_policy.cpp src/instance_guard.cpp)
    target_include_directories(${target} PUBLIC src)
    target_link_libraries(${target} PUBLIC lock_auth)
    target_compile_definitions(${target} PUBLIC MAGI_LOCK_DEVELOPMENT_UNLOCK=${development} MAGI_LOCK_FAILURE_INJECTION=$<AND:$<BOOL:${MAGI_LOCK_FAILURE_INJECTION}>,$<BOOL:${development}>>)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endfunction()
if(MAGI_LOCK_DEVELOPMENT_UNLOCK)
    message(WARNING "UNSAFE DEVELOPMENT BUILD: explicitly enabled timed unlock; never deploy")
    lock_core(lock_core 1)
else()
    lock_core(lock_core 0)
endif()

    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/failure-fixtures")

        file(CREATE_LINK "${MAGI_LOCK_AUTH_FIXTURE}" "${CMAKE_CURRENT_BINARY_DIR}/failure-fixtures/fixture-${mode}" SYMBOLIC)
    endforeach()

    find_program(CARGO cargo REQUIRED)
    file(GLOB_RECURSE AUTH_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/auth/src/*.rs")
    set(AUTH_TARGET "${CMAKE_CURRENT_BINARY_DIR}/cargo")
    set(AUTH_BINARY "${AUTH_TARGET}/release/magi-lock-auth")
    add_custom_command(OUTPUT "${AUTH_BINARY}"
        COMMAND "${CMAKE_COMMAND}" -E env
            --unset=CARGO_ENCODED_RUSTFLAGS
            "RUSTFLAGS=-C relocation-model=pic -C link-arg=-Wl,-z,relro,-z,now,-z,noexecstack"
            "${CARGO}" build --release --locked --offline --bin magi-lock-auth
            --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/auth/Cargo.toml" --target-dir "${AUTH_TARGET}"
        DEPENDS ${AUTH_SOURCES} auth/Cargo.toml auth/Cargo.lock VERBATIM)
    add_custom_target(production_auth ALL DEPENDS "${AUTH_BINARY}")
    # Fixed FHS layout. No activation, daemon reload, config migration or hooks.
    install(TARGETS magi-lock-proof RUNTIME DESTINATION lib/magi)
    install(PROGRAMS "${AUTH_BINARY}" DESTINATION lib/magi)
    install(PROGRAMS integration/magi-lock-request DESTINATION bin)
    install(FILES integration/magi-lock@.service DESTINATION lib/systemd/user)
    install(FILES integration/pam/magi-lock DESTINATION /etc/pam.d)
endif()
